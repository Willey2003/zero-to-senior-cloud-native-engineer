  function read() {
    var s = {};
    root.querySelectorAll('[data-k]').forEach(function (el) { s[el.getAttribute('data-k')] = el.value; });

  function plan(s) {
    var ip = function (n) { return s.net + '.' + n; };
    var p = { vip: ip(20), cps: [], etcds: [], workers: [] };
    for (var i = 0; i < s.cp; i++) p.cps.push({ name: i === 0 ? 'manager' : 'manager-' + (i + 1), ip: ip(i === 0 ? 14 : 20 + i) });
    if (s.topo === 'external') for (var e = 0; e < s.etcd; e++) p.etcds.push({ name: 'etcd-' + (e + 1), ip: ip([12, 13, 19, 33, 34, 35, 36][e]) });
    else p.etcds = p.cps.map(function (c) { return { name: c.name, ip: c.ip }; });
    for (var w = 0; w < s.w; w++) p.workers.push({ name: 'worker-' + String.fromCharCode(97 + (w % 26)) + (w >= 26 ? w : ''), ip: ip(100 + w) });
    p.lbIp = s.lb === 'keepalived' ? ip(20) : s.lb === 'external' ? 'lb.corp.example' : ip(20);
    return p;
  }

  function maths(s) {
    var n = s.etcd, quorum = Math.floor(n / 2) + 1, tol = Math.floor((n - 1) / 2);
    var cpTol = s.topo === 'stacked' ? Math.min(tol, s.cp - 1) : s.cp - 1;
    return { quorum: quorum, etcdTol: tol, cpTol: Math.max(cpTol, 0), vms: s.cp + s.w + (s.topo === 'external' ? s.etcd : 0) + (s.lb === 'keepalived' ? 2 : 0) };
  }

  function notes(s, m) {
    var out = [];
    if (s.cp === 1) out.push(['bad', 'Single control plane: the API server is a single point of failure. A VIP still helps, because it lets you add managers later without re-pointing every kubelet.']);
    if (s.etcd % 2 === 0) out.push(['warn', s.etcd + ' etcd members tolerate the same number of failures as ' + (s.etcd - 1) + ', yet add one more voter to every write. Use an odd count.']);
    if (s.etcd > 5) out.push(['warn', s.etcd + ' etcd members add write latency (every write needs ' + m.quorum + ' acks). 5 is the practical ceiling for most clusters.']);
    if (s.topo === 'stacked' && s.cp >= 3) out.push(['ok', 'Stacked etcd: fewer VMs, but losing a control-plane node also removes an etcd voter. Tolerates ' + m.etcdTol + ' node loss(es).']);
    if (s.topo === 'external' && s.etcd >= 3) out.push(['ok', 'External etcd: control-plane and data-plane failures are decoupled. The API survives while 1 manager and ' + m.quorum + ' etcd members are up.']);
    if (s.lb === 'kubevip') out.push(['ok', 'kube-vip in ARP mode: one leader holds the VIP via a Lease; failover is a gratuitous ARP away (measured ~1 s in the lab). All managers must share one L2 segment.']);
    if (s.lb === 'keepalived') out.push(['ok', 'HAProxy + keepalived on 2 extra VMs: real health-checked TCP load balancing across all API servers, at the cost of two more machines to patch.']);
    if (s.lb === 'external') out.push(['ok', 'External LB: point it at :6443 on every manager with a TCP health check on /livez, and put its DNS name in certSANs.']);
    if (s.w < 2) out.push(['warn', 'One worker: any workload with replicas > 1 and anti-affinity cannot schedule. Plan for at least 2-3.']);
    return out;
  }

  var X = function (v) { return Math.round(v); };

    return s + '</g>';
  }
  function row(items, y, W, maxShow, boxW) {
    var shown = items.slice(0, maxShow), extra = items.length - shown.length;
    var n = shown.length + (extra > 0 ? 1 : 0), gap = 12;
    var bw = Math.min(boxW, (W - 40 - gap * (n - 1)) / n), total = n * bw + (n - 1) * gap, x0 = (W - total) / 2;
    return { shown: shown, extra: extra, bw: bw, xs: shown.map(function (_, i) { return x0 + i * (bw + gap); }), xExtra: x0 + shown.length * (bw + gap), y: y };
  }
  function link(x1, y1, x2, y2, color, dash) {
    return '<path d="M' + X(x1) + ' ' + y1 + ' C' + X(x1) + ' ' + ((y1 + y2) / 2) + ' ' + X(x2) + ' ' + ((y1 + y2) / 2) + ' ' + X(x2) + ' ' + y2 + '" stroke="' + color + '" stroke-width="1.2" fill="none" opacity=".55"' + (dash ? ' stroke-dasharray="4 4"' : '') + '/>';
  }
  function label(x, y, t) { return '<text x="' + x + '" y="' + y + '" fill="' + C.dim + '" font-size="10" letter-spacing="1.5">' + t + '</text>'; }

    out += label(16, yLb + 8, 'API ENDPOINT :6443');
    out += nodeBox(lbX, yLb, lbW, 46, C.lb, lbName, s.lb === 'external' ? p.lbIp + ':6443' : p.vip + ':6443');
    out += label(16, yCp - 8, 'CONTROL PLANE');
    cpRow.xs.forEach(function (x, i) { out += nodeBox(x, yCp, cpRow.bw, ext ? 50 : 60, C.cp, cpRow.shown[i].name, cpRow.shown[i].ip, !ext); });
    if (ext) {
      out += label(16, yEtcd - 8, 'ETCD (' + s.etcd + ' MEMBERS, QUORUM ' + (Math.floor(s.etcd / 2) + 1) + ')');

    if (wRow.extra > 0) out += nodeBox(wRow.xExtra, yW, wRow.bw, 44, C.dim, '+' + wRow.extra + ' more', 'workers');

  function kubeadmCfg(s, p) {
    var ep = s.lb === 'external' ? p.lbIp : p.vip;
    var sans = [ep].concat(p.cps.map(function (c) { return c.ip; }), p.cps.map(function (c) { return c.name; }));
    var y = [
      '# kubeadm-config.yaml  (kubeadm init --config kubeadm-config.yaml --upload-certs)',
      'apiVersion: kubeadm.k8s.io/v1beta4',
      'kind: InitConfiguration',
      'localAPIEndpoint:',
      '  advertiseAddress: ' + p.cps[0].ip,
      '  bindPort: 6443',
      'nodeRegistration:',
      '  name: ' + p.cps[0].name,
      '  criSocket: unix:///var/run/crio/crio.sock',
      '---',
      'apiVersion: kubeadm.k8s.io/v1beta4',
      'kind: ClusterConfiguration',
      'kubernetesVersion: v1.32.13',
      'controlPlaneEndpoint: ' + ep + ':6443',
      'apiServer:',
      '  certSANs:'
    ].concat(sans.map(function (x) { return '  - ' + x; }));
    if (s.topo === 'external') {
      y = y.concat(['etcd:', '  external:', '    endpoints:'], p.etcds.map(function (e) { return '    - https://' + e.ip + ':2379'; }),
        ['    caFile: /etc/kubernetes/pki/etcd/ca.crt', '    certFile: /etc/kubernetes/pki/apiserver-etcd-client.crt', '    keyFile: /etc/kubernetes/pki/apiserver-etcd-client.key']);
    } else {
      y = y.concat(['etcd:', '  local:', '    dataDir: /var/lib/etcd', '    # stacked: one etcd member per control-plane node (' + s.cp + ' total)']);
    }
    y = y.concat(['networking:', '  podSubnet: 192.168.0.0/16   # Calico default', '  serviceSubnet: 10.96.0.0/12']);
    return y.join('\n');
  }

  function endpointCfg(s, p) {
    if (s.lb === 'kubevip') {
      return [
        '# /etc/kubernetes/manifests/kube-vip.yaml  (one per manager; only vip_interface may differ)',
        '# generate: kube-vip manifest pod --interface ens34 --address ' + p.vip + ' --controlplane --arp --leaderElection',
        'env:',
        '- {name: vip_arp,             value: "true"}',
        '- {name: port,                value: "6443"}',
        '- {name: vip_interface,       value: ens34}',
        '- {name: vip_subnet,          value: "32"}',
        '- {name: cp_enable,           value: "true"}',
        '- {name: vip_leaderelection,  value: "true"}',
        '- {name: vip_leasename,       value: plndr-cp-lock}',
        '- {name: vip_leaseduration,   value: "15"}',
        '- {name: vip_renewdeadline,   value: "10"}',
        '- {name: vip_retryperiod,     value: "2"}',
        '- {name: address,             value: ' + p.vip + '}',
        'image: ghcr.io/kube-vip/kube-vip:v1.2.4',
        'securityContext: {capabilities: {add: [NET_ADMIN, NET_RAW], drop: [ALL]}}',
        'hostNetwork: true'
      ].join('\n');
    }
    if (s.lb === 'keepalived') {
      return ['# /etc/haproxy/haproxy.cfg  (on both LB VMs; keepalived floats ' + p.vip + ')',
        'frontend k8s-api',
        '    bind ' + p.vip + ':6443',
        '    mode tcp',
        '    option tcplog',
        '    default_backend k8s-api',
        '',
        'backend k8s-api',
        '    mode tcp',
        '    balance roundrobin',
        '    option httpchk GET /livez',
        '    http-check expect status 200',
        '    default-server check check-ssl verify none inter 3s fall 3 rise 2'
      ].concat(p.cps.map(function (c) { return '    server ' + c.name + ' ' + c.ip + ':6443'; })).join('\n');
    }
    return ['# External load balancer target group', '# listener: TCP 6443  ->  targets:'].concat(p.cps.map(function (c) { return '#   ' + c.ip + ':6443  (' + c.name + ')'; }),
      ['# health check: HTTPS GET /livez  every 5s, unhealthy after 3', '# add the LB DNS name to apiServer.certSANs before kubeadm init']).join('\n');
  }

  function joinCmds(s, p) {
    var ep = s.lb === 'external' ? p.lbIp : p.vip;
    var out = ['# on ' + p.cps[0].name + ' after kubeadm init --upload-certs:', 'kubeadm token create --print-join-command', 'kubeadm init phase upload-certs --upload-certs   # prints <cert-key>, valid 2h', ''];
    p.cps.slice(1).forEach(function (c) {
      out.push('# ' + c.name);
      out.push('kubeadm join ' + ep + ':6443 --token <token> \\');
      out.push('  --discovery-token-ca-cert-hash sha256:<hash> \\');
      out.push('  --control-plane --certificate-key <cert-key> \\');
      out.push('  --apiserver-advertise-address ' + c.ip);
      out.push('');
    });
    out.push('# every worker');
    out.push('kubeadm join ' + ep + ':6443 --token <token> --discovery-token-ca-cert-hash sha256:<hash>');
    return out.join('\n');
  }

  function ports(s) {
    var l = ['# ufw rules (default deny incoming), mirroring 02-ha-control-plane/base-setup.sh', 'LAN=' + read().net + '.0/24'];
    l.push('for p in 6443 10250 10257 10259; do ufw allow from "$LAN" to any port "$p" proto tcp; done   # control plane');
    if (s.topo === 'stacked') l.push('ufw allow from "$LAN" to any port 2379:2380 proto tcp   # stacked etcd client + peer');
    else l.push('# on etcd hosts: ufw allow from "$LAN" to any port 2379:2380 proto tcp');
    l.push('ufw allow from "$LAN" to any port 179  proto tcp   # calico bgp');
    l.push('ufw allow from "$LAN" to any port 5473 proto tcp   # calico typha');
    l.push('ufw allow from "$LAN" to any port 4789 proto udp   # calico vxlan');
    l.push('# workers: 10250/tcp from LAN, 30000:32767/tcp for NodePorts');
    if (s.lb === 'kubevip') l.push('ufw allow from "$LAN" to any port 2112 proto tcp   # kube-vip prometheus (optional)');
    return l.join('\n');
  }

  var TABS = [['kubeadm', 'kubeadm-config.yaml'], ['endpoint', 'API endpoint'], ['join', 'Join commands'], ['ports', 'Firewall']];

  function render() {
    var s = read(), p = plan(s), m = maths(s);
    var etcdSel = q('[data-k=etcd]');
    etcdSel.disabled = s.topo === 'stacked';
    if (s.topo === 'stacked') etcdSel.value = String(s.cp);
    q('[data-o=w]').textContent = s.w;
    q('[data-o=stats]').innerHTML = [
      [m.quorum + ' / ' + s.etcd, 'etcd quorum'],
      [m.etcdTol, 'etcd failures tolerated'],
      [m.cpTol, 'manager failures tolerated'],
      [m.vms, 'VMs to provision']
    ].map(function (r) { return '<div class="kb-stat"><b style="color:' + (r[1].indexOf('tolerated') > -1 && r[0] === 0 ? C.bad : C.text) + '">' + r[0] + '</b><span>' + r[1] + '</span></div>'; }).join('');
    q('[data-o=svg]').innerHTML = svg(s, p);
    var col = { ok: C.ok, warn: C.warn, bad: C.bad };
    q('[data-o=notes]').innerHTML = notes(s, m).map(function (n) { return '<li style="border-color:' + col[n[0]] + '">' + esc(n[1]) + '</li>'; }).join('');
    q('[data-o=tabs]').innerHTML = TABS.map(function (t) { return '<button type="button" role="tab" data-tab="' + t[0] + '" aria-selected="' + (t[0] === activeTab) + '">' + t[1] + '</button>'; }).join('');
    var code = activeTab === 'kubeadm' ? kubeadmCfg(s, p) : activeTab === 'endpoint' ? endpointCfg(s, p) : activeTab === 'join' ? joinCmds(s, p) : ports(s);
    q('[data-o=code]').textContent = code;
  }

  root.addEventListener('input', render);
  root.addEventListener('change', render);
  root.addEventListener('click', function (e) {
    var t = e.target.closest('[data-tab]');
    if (t) { activeTab = t.getAttribute('data-tab'); render(); return; }
    if (e.target.classList.contains('kb-copy')) {
      var txt = q('[data-o=code]').textContent, b = e.target;
      var done = function () { b.textContent = 'Copied'; setTimeout(function () { b.textContent = 'Copy'; }, 1400); };
      if (navigator.clipboard) navigator.clipboard.writeText(txt).then(done, function () {});
    }
  });
  render();

The control plane is the nervous system of your Kubernetes ecosystem. In a production cluster, achieving true High Availability (HA) demands eliminating every single single point of failure (SPOF). This masterclass deconstructs the architecture required to scale control plane structures, manage redundant database states, and handle traffic load balancing.

## 1. Control Plane Architecture Selection

When engineering a highly available Kubernetes management layer, you must choose between two distinct structural designs for your backing storage (`etcd`):

### Stacked Control Plane Nodes
In this topology, each node running control plane services (the API server, controller manager, and scheduler) also runs a local co-located `etcd` pod. 
- **The Advantages:** Simple to spin up, requires fewer individual servers, and scales linearly.
- **The Tradeoff:** If a node experiences an infrastructure drop, you lose both a control plane api worker and a member of your consensus cluster simultaneously.

### External Dedicated etcd Clusters
This decoupled strategy extracts the database layer entirely, hosting `etcd` members on independent hardware nodes.
- **The Advantages:** Complete isolation of concerns. Heavy compute spikes on API workloads will not throttle state store operations.
- **The Tradeoff:** Requires double the machine footprint and demands meticulous certificate routing management.

## 3. High Availability Load Balancing Configuration

The API servers sitting across your control nodes must sit behind an enterprise-grade TCP load balancer. Below is a production-hardened `haproxy.cfg` blueprint designed to handle traffic parsing with health checks:

```haproxy
global
    log /dev/log local0
    group haproxy
    daemon

defaults
    log     global
    mode    http
    timeout connect 10s
    timeout client  60s
    timeout server  60s

frontend k8s_api_frontend
    bind *:6443
    mode tcp
    option tcplog
    default_backend k8s_api_backends

backend k8s_api_backends
    mode tcp
    option tcp-check
    balance roundrobin
    default-server inter 10s downinter 5s rise 2 fall 3 slowstart 60s maxconn 250 maxqueue 256 weight 100
    server cp-node-01 10.0.10.11:6443 check
    server cp-node-02 10.0.10.12:6443 check
    server cp-node-03 10.0.10.13:6443 check
```
